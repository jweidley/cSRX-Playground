set interfaces ge-0/0/0 description trust
set interfaces ge-0/0/1 unit 0 family inet address 172.16.30.2/24
set interfaces ge-0/0/1 description untrust

set security nat source rule-set trust-to-untrust to zone untrust

set security nat source rule-set trust-to-untrust rule snat1 then source-nat interface

set security nat source rule-set noc-to-untrust rule snat3 then source-nat interface
